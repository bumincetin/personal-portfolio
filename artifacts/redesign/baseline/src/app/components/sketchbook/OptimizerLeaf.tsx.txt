'use client';

import { useState } from 'react';
import type { Locale } from '@/lib/translations';
import { getLibraryUI } from '@/lib/content/library-ui';

/** One instrument, mounted only after explicit activation; never in live clones. */
export default function OptimizerLeaf({ locale }: { locale: Locale }) {
  const copy = getLibraryUI(locale);
  const [Instrument, setInstrument] = useState<typeof import('@/app/components/optimizer/PortfolioOptimizer').default | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const open = async () => {
    setLoading(true); setError('');
    try { const loaded = await import('@/app/components/optimizer/PortfolioOptimizer'); setInstrument(() => loaded.default); }
    catch { setError(copy.optimizerFailed); }
    finally { setLoading(false); }
  };
  return Instrument ? <Instrument/> : <><button type="button" data-open-optimizer className="reader-instrument-button" aria-busy={loading} disabled={loading} onClick={open}>{copy.launchOptimizer}</button><p role="status">{error}</p></>;
}
